1. LOG IN TO DOCKER HUB

   sudo su (Run this as sudo)
   docker login -u numurusinc -p #<password-or-token>

   If you have a Personal Access Token, use that instead of the password — it's safer and can be revoked without changing the account password. You can generate one at hub.docker.com → Account Settings → Personal Access Tokens (set scope to Read & Write).


2. SET UP BUILDX BUILDER (one-time setup)


   #docker buildx create --name nepi_builder --use

   #docker buildx inspect --bootstrap


   You only need to do this once per machine.


3. UPDATE INFO
   nepistart
   sleep 5
   nepiupdate
   source ${DCFG}/load_docker_config.sh


3. SETUP CORRECT PLATFORM

########################
if is_valid_amd64; then
   if [[ "$NEPI_RUNNING_TAG" == *"$cuda"* ]]; then
         hub_tag="latest-amd64-cuda"
   else
         hub_tag="latest-amd64"
   fi
   platform=linux/amd64

elif is_valid_jetson; then
   hub_tag=="latest-jetson"
   platform=linux/arm64

elif is_valid_arm64; then
   hub_tag="latest-arm64"
   platform=linux/arm64
fi

   
4. PUSH
   nepiupdate
   source ${DCFG}/load_docker_config.sh
   nepistop
   sudo docker tag ${NEPI_RUNNING_FS}:${NEPI_RUNNING_TAG} numurusinc/nepi:${hub_tag}
   echo "Pushing ${NEPI_RUNNING_FS}:${NEPI_RUNNING_TAG} to docker hub tag ${hub_tag}"
   docker push numurusinc/nepi:${hub_tag}
   sudo docker rmi numurusinc/nepi:${hub_tag}
